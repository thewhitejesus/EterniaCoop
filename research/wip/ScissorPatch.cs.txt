// The game's renderer clips everything GameNative draws to the original picture (a scissor rectangle, e.g. 2560x1011
// from y 174 on a 2560x1440 screen). Zoomed out past the game's own limit, the battle view reaches below that picture,
// so the extended ground (BattleGround) would be cut off there and the stage's clear colour (black, green, teal...)
// shows instead. While the co-op camera has zoomed out that far in battle, the full-width picture scissor is switched
// off right after the game (and any other mod, e.g. Uncropped Backgrounds, which widens it) has set it. Narrower
// scissors (menu lists and the like) are left alone.
using HarmonyLib;
using UnityEngine;
using UnityEngine.Rendering;

namespace EterniaCoop
{
    [HarmonyPatch("UgxGraphicsGlobalStateManager, Assembly-CSharp", "ApplyScissorTest")]
    static class ScissorPatch
    {
        const short ZoomGameMax = 0x29C;
#if COOP_DEBUG
        static readonly System.Collections.Generic.HashSet<string> seen = new System.Collections.Generic.HashSet<string>();
#endif

        [HarmonyPriority(Priority.Last)]
        static void Postfix(CommandBuffer commandBuffer, bool isEnable, int scissorLeft, int scissorTop, int scissorWidth, int scissorHeight)
        {
#if COOP_DEBUG
            string key = isEnable + " " + scissorLeft + "," + scissorTop + " " + scissorWidth + "x" + scissorHeight;
            if (seen.Count < 40 && seen.Add(key)) Plugin.Log.LogInfo("Scissor: " + key + " (screen " + Screen.width + "x" + Screen.height + ")");
#endif
            if (commandBuffer == null || !isEnable || scissorWidth < Screen.width * 0.9f || !Active()) return;
            commandBuffer.DisableScissorRect();
        }

        static bool Active()
        {
            return Settings.CoopCamera.Value && Features.Camera && Native.Base != 0 && Battle.Actors() != System.IntPtr.Zero &&
                System.Runtime.InteropServices.Marshal.ReadInt16(Native.At(0x12CEA9C)) > ZoomGameMax;
        }
    }
}
